public PairNode( String k, Integer v ){
        this.key=k;
        this.value=v;
        this.next=null;
    }

private int hashFunction( String key ){
        String s=key;
        int sum=0;
        if(s.length()%2!=0) s+='N';
        for(int i=0;i<s.length()-1;i+=2){
            String x=Integer.toString((int)s.charAt(i))+Integer.toString((int)s.charAt(i+1));
            sum+=Integer.parseInt(x);
        }
        return sum%ht.length;
    }

public String searchHashtable( Object[] keyValuePair ){
        String key=(String)keyValuePair[0];
        int index=hashFunction(key);
        PairNode current=ht[index];
        while(current!=null){
            if(current.key.equals(key)){
                return "Found";
            }
            current=current.next;
        }
        return "Not Found";
    }

public FruitNode(String name, Integer price){
        fruit[0]=name;
        fruit[1]=price;
        next=null;
    }

private int hashFunction( String key ){
        String s=key;
        int sum=0;
        if(s.length()%2==0){
            for(int i=0;i<s.length();i+=2){
                sum+=(int)s.charAt(i);
            }
        }else{
            for(int i=1;i<s.length();i+=2){
                sum+=(int)s.charAt(i);
            }
        }
        return sum%ht.length;
    }

public void insert(String key, Integer value){
        int index=hashFunction(key);
        FruitNode current=ht[index];
        while(current!=null){
            if(current.fruit[0].equals(key)){
                current.fruit[1]=value;
                return;
            }
            current=current.next;
        }
        FruitNode newNode=new FruitNode(key,value);
        newNode.next=ht[index];
        ht[index]=newNode;
    }

public PairNode( Integer k, String v ){
        key=k;
        value=v;
        next=null;
    }

private int hashFunction( Integer key ){
        return (key+3)%ht.length;
    }

public void remove( Integer key ){
        Integer index=hashFunction(key);
        PairNode current=ht[index];
        PairNode prev=null;
        while(current!=null){
            if(current.key.equals(key)){
                if(prev==null){
                    ht[index]=current.next;
                }else{
                    prev.next=current.next;
                }
                return;
            }
            prev=current;
            current=current.next;
        }
    }

public static void removeBlock(Stack stack, int n) {
        Stack stk=new Stack();
        while(!stack.isEmpty()){
            stk.push(stack.pop());
        }
        for(int i=1;i<n;i++){
            stack.push(stk.pop());
        }
        stk.pop();
        while(!stk.isEmpty()){
            stack.push(stk.pop());
        }
    }

public static Stack conditionalReverse(Stack stack) {
        Stack stk=new Stack();
        stk.push(stack.pop());
        while(!stack.isEmpty()){
            if(!(stack.peek().equals(stk.peek()))){
                stk.push(stack.pop());
            }else{
                stack.pop();
            }
        }
        return stk;
    }

public void enqueueCall(int customerId, boolean isVip) {
        if(isVip){
            this.vipQueue.enqueue(customerId);
            System.out.printf("Customer %d added to VIP queue.\n",customerId);
        }else{
            this.regularQueue.enqueue(customerId);
            System.out.printf("Customer %d added to Regular queue.\n",customerId);
        }
    }

public void dequeueCall() {
        if(vipQueue.isEmpty()&&regularQueue.isEmpty()){
            System.out.println("No calls in the queue.");
            return;
        } 
        if(!vipQueue.isEmpty()){
            System.out.printf("Processing VIP Customer %d.\n",vipQueue.dequeue());
        }else{
            System.out.printf("Processing Regular Customer %d.\n",regularQueue.dequeue());
        }
    }
