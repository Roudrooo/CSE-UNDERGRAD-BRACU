// Task01
public static BTNode convertMirror( BTNode root ){
        if(root!=null){
            BTNode left=root.left,right=root.right;
            root.left=right;
            root.right=left;
            convertMirror(left); 
            convertMirror(right);
            return root;
        }
        return null;
    }


// Task02
public static void smallestLevel( BTNode root, Integer[] levelArray, int lvl ){
        if(root!=null){
            if(levelArray[lvl]==null) levelArray[lvl]=(Integer)root.elem;
            else if((Integer)root.elem<levelArray[lvl]) levelArray[lvl]=(Integer)root.elem;
            smallestLevel(root.left,levelArray,lvl+1);
            smallestLevel(root.right,levelArray,lvl+1);
        }
        
    }


// Task03
public static Integer sumTree( BTNode root ){
            return helperModSum(root,0);
        }
        public static Integer helperModSum(BTNode root,Integer level){
            if(root==null) return 0;
            Integer sum;
            if(level==0) sum=(Integer)root.elem;
            else sum=(Integer)root.elem%level;
            return sum+helperModSum(root.left,level+1)+helperModSum(root.right,level+1);
        }


// Task04
public static void swapChild( BTNode root, int lvl, int M ){
        if(root!=null&&M!=0){
            BTNode left=root.left,right=root.right;
            root.left=right;
            root.right=left;
            swapChild(root.left,lvl+1,M-1);
            swapChild(root.right,lvl+1,M-1);
        }
    }


// Task05
public static Integer subtractSummation( BTNode root ){
        return subTreeSum(root.left)-subTreeSum(root.right);
    }
    public static Integer subTreeSum(BTNode root){
        if(root!=null) return (Integer)root.elem+subTreeSum(root.left)+subTreeSum(root.right);
        return 0;
    }


// Task06
public static Integer levelSum( BTNode root ){
        return signSum(root,0);
    }
    public static Integer signSum(BTNode root, int lvl){
        if(root!=null){
            Integer sign;
            if(lvl%2==0) sign=-1;
            else sign=1;
            return sign*(Integer)root.elem+signSum(root.left,lvl+1)+signSum(root.right,lvl+1);
        }
        return 0;
    }
